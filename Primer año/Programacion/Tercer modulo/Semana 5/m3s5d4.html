🎯 Competencia General

Aplica las estructuras condicionales if, else, else if y los operadores de comparación y lógicos en Dart para crear un programa interactivo que tome decisiones basadas en la entrada del usuario, consolidando los conceptos de control de flujo.

📖 Conceptuales (Saber)

  • Comprende el funcionamiento de if, else y else if.
  • Reconoce los operadores de comparación y lógicos.
  • Entiende el flujo de ejecución de condiciones anidadas.
  • Identifica cuándo usar cada estructura condicional.

🛠️ Procedimentales (Saber hacer)

  • Leer datos del usuario con stdin.readLineSync().
  • Evaluar condiciones con if, else y else if.
  • Combinar condiciones con && y ||.
  • Mostrar resultados usando interpolación de Strings.

if (edad < 18) {
  clasificacion = "Eres menor de edad";

  clasificacion = "Eres un adulto joven";
} else if (edad > 30 && edad <= 60) {
  clasificacion = "Eres un adulto";
} else {
  clasificacion = "Eres una persona mayor";

🔹 Paso 4: Clasificar por ingresos

String nivelIngresos = "";

🔹 Código completo del proyecto

import 'dart:io';

void main() {
  print("Ingresa tu edad:");
  int edad = int.parse(stdin.readLineSync()!);

  print("Ingresa tus ingresos mensuales (en USD):");
  double ingresos = double.parse(stdin.readLineSync()!);

  print("Ingresa tu nivel de estudios (Bachillerato, Universitario, Posgrado):");
  String estudios = stdin.readLineSync()!;

  String clasificacion = "";
  if (edad < 18) {
    clasificacion = "Eres menor de edad";
  } else if (edad >= 18 && edad <= 30) {
    clasificacion = "Eres un adulto joven";
  } else if (edad > 30 && edad <= 60) {
    clasificacion = "Eres un adulto";
  } else {
    clasificacion = "Eres una persona mayor";
  }

  String nivelIngresos = "";
  if (ingresos < 1000) {
    nivelIngresos = "ingresos bajos";
  } else if (ingresos >= 1000 && ingresos < 3000) {
    nivelIngresos = "ingresos moderados";
  } else if (ingresos >= 3000 && ingresos < 6000) {
    nivelIngresos = "ingresos altos";
  } else {
    nivelIngresos = "ingresos muy altos";
  }

  String mensaje = "";
  if (estudios == "Bachillerato" || estudios == "bachillerato") {
    mensaje = "Tienes estudios de bachillerato";
  } else if (estudios == "Universitario" || estudios == "universitario") {
    mensaje = "Tienes estudios universitarios";
  } else if (estudios == "Posgrado" || estudios == "posgrado") {
    mensaje = "Tienes estudios de posgrado";
  } else {
    mensaje = "Nivel de estudios no reconocido";
  }

  print("$clasificacion, con $nivelIngresos. $mensaje.");
}
